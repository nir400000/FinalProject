import struct

def _create_wav_header(sample_rate: int, channels: int) -> bytes:
    """Generate a standard 44-byte WAV header for a continuous stream."""
    bits_per_sample = 16
    byte_rate = sample_rate * channels * (bits_per_sample // 8)
    block_align = channels * (bits_per_sample // 8)
    
    # We use a massive file size (0x7FFFFFFF) because it's an endless stream.
    header = struct.pack(
        '<4sI4s4sIHHIIHH4sI',
        b'RIFF',
        0x7FFFFFFF,  # Chunk size
        b'WAVE',
        b'fmt ',
        16,          # Subchunk1 size (16 for PCM)
        1,           # Audio format (1 for PCM)
        channels,
        sample_rate,
        byte_rate,
        block_align,
        bits_per_sample,
        b'data',
        0x7FFFFFFF   # Subchunk2 size
    )
    return header
